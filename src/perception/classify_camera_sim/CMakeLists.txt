cmake_minimum_required(VERSION 3.8)
project(classify_camera_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core)
add_executable(align_rgbd src/align_rgbd.cpp)
ament_target_dependencies(align_rgbd rclcpp sensor_msgs cv_bridge message_filters)
target_link_libraries(align_rgbd ${OpenCV_LIBS})
target_compile_features(align_rgbd PUBLIC cxx_std_17)
install(TARGETS align_rgbd DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_registration test/test_registration.cpp)
  target_include_directories(test_registration PRIVATE src ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(test_registration ${OpenCV_LIBS})
  target_compile_features(test_registration PUBLIC cxx_std_17)
endif()
ament_package()
