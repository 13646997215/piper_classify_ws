cmake_minimum_required(VERSION 3.8)
project(classify_task)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(classify_interfaces REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
add_executable(classify_task_node src/classify_task.cpp)
target_compile_features(classify_task_node PUBLIC cxx_std_17)
target_compile_options(classify_task_node PRIVATE -Wall -Wextra)
ament_target_dependencies(classify_task_node rclcpp std_srvs std_msgs sensor_msgs gazebo_msgs
  visualization_msgs classify_interfaces moveit_task_constructor_core moveit_ros_planning_interface)
install(TARGETS classify_task_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(Eigen3 REQUIRED)
  ament_add_gtest(test_bin_layout test/test_bin_layout.cpp)
  target_include_directories(test_bin_layout PRIVATE src)
  target_link_libraries(test_bin_layout Eigen3::Eigen)
  target_compile_features(test_bin_layout PUBLIC cxx_std_17)
endif()
ament_package()
